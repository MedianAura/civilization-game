Create `src/scenes/PreloadScene.ts` (per D-06 — loads assets via Phaser.Loader, shows progress bar; Phase 1 has no real assets to load yet, so the progress bar wires to Phaser's loader lifecycle events even with an empty load queue):
```typescript
import Phaser from "phaser";

export class PreloadScene extends Phaser.Scene {
  private progressBar!: Phaser.GameObjects.Graphics;
  private progressBox!: Phaser.GameObjects.Graphics;

  constructor() {
    super("PreloadScene");
  }

  preload(): void {
    const { width, height } = this.cameras.main;

    this.progressBox = this.add.graphics();
    this.progressBox.fillStyle(0x222222, 0.8);
    this.progressBox.fillRect(width / 2 - 160, height / 2 - 25, 320, 50);

    this.progressBar = this.add.graphics();

    this.load.on(Phaser.Loader.Events.PROGRESS, (value: number) => {
      this.progressBar.clear();
      this.progressBar.fillStyle(0xffffff, 1);
      this.progressBar.fillRect(width / 2 - 150, height / 2 - 15, 300 * value, 30);
    });

    this.load.on(Phaser.Loader.Events.COMPLETE, () => {
      this.progressBar.destroy();
      this.progressBox.destroy();
    });
  }

  create(): void {
    this.scene.start("GameScene");
  }
}
```
